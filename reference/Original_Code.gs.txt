function posValidateIds_(env,r,previous){

  // Existing linked services cannot be deleted, retyped or assigned a new ID. This avoids orphaning production records.
  for(const prior of old){if(!r.items.some(x=>x.kind===prior.kind&&x.existingId===prior.existingId))throw Error('A linked service ID cannot be changed or removed. Edit its details instead.');}
  const dup=new Set();for(const item of r.items){const id=String(item.existingId||'').toUpperCase(),key=item.kind+'|'+id;if(dup.has(key))throw Error('Duplicate item ID '+id);dup.add(key);item.existingId=id;

      const pattern=item.kind==='editing'?/^RF\d{5}[A-Z]$/:/^[A-Z]{2}\d[A-Z]\d$/;
      if(!pattern.test(id))throw Error('Invalid '+item.kind+' ID '+id);

        const tracking=posSheet_(POS_MASTER_ID,'Tracking');
        if(posFindInCol_(tracking,1,id,2)>0||posFindInCol_(env.remit,17,id,6)>0)throw Error('Editing ID already exists: '+id);

/** Receiver name belongs to shipping details, not the payer/remittance customer. Prefer the shared shipping receiver; retain legacy per-item receiver fallback. */
function posFrameReceiverParts_(item,r){
  const full=String(((r.shipping||{}).receiverName||item.receiverName||'')).trim().replace(/\s+/g,' ').slice(0,160);
  if(!full)return {first:String(r.first||''),last:String(r.last||'')};
  if(/^n\/?a$/i.test(full))return {first:'N/A',last:''};
  const words=full.split(' ');
  // Match the Admin POS Full Name convention: final word as surname.
  // One-word recipient names are accepted without creating a false last name.
  return words.length===1?{first:words[0],last:''}:{first:words.slice(0,-1).join(' '),last:words[words.length-1]};
}
/** V16.14: Strict frame-row resolver. An existing linked frame must NEVER create a second Orders row.
 * Key: AN (POS Item Key, col 40); definitive order ID: C (col 3).
 * A stale/missing key is repaired only when the original POS transaction owns the same row.
 */
function posResolveFrameRow_(sh,r,item,previous){
  const id=String(item.existingId||'').trim().toUpperCase();
  const key=posItemKey_(r,item);
  if(!id)throw Error('Missing Frame Order ID; nothing written.');
  const last=sh.getLastRow();
  const matchesKey=[],matchesId=[];
  if(last>=2){
    const n=last-1;
    const ids=sh.getRange(2,3,n,1).getDisplayValues();
    const keys=sh.getRange(2,40,n,1).getDisplayValues();
    for(let i=0;i<n;i++){
      if(String(ids[i][0]||'').trim().toUpperCase()===id)matchesId.push(i+2);
      if(String(keys[i][0]||'').trim().toUpperCase()===key.toUpperCase())matchesKey.push(i+2);
    }
  }
  if(matchesId.length>1||matchesKey.length>1)
    throw Error('Duplicate Frame Order rows detected for '+id+'. Save stopped; review Orders rows before editing.');
  const keyRow=matchesKey[0]||0, idRow=matchesId[0]||0;
  if(keyRow&&idRow&&keyRow!==idRow)
    throw Error('Frame Order '+id+' has conflicting POS keys in Orders. No rows changed.');
  if(keyRow&&!idRow)
    throw Error('POS Item Key exists but Frame Order ID differs for '+id+'. No rows changed.');
  const oldItems=Array.isArray(previous&&previous.items)?previous.items:[];
  const wasLinked=oldItems.some(x=>x.kind==='frame'&&String(x.existingId||'').trim().toUpperCase()===id);
  const found=keyRow||idRow;
  if(found){
    const owner=String(sh.getRange(found,39).getDisplayValue()||'').trim();
    const currentKey=String(sh.getRange(found,40).getDisplayValue()||'').trim();
    if(owner&&owner!==String(r.posId))
      throw Error('Frame Order '+id+' belongs to a different POS transaction. No rows changed.');
    if(currentKey&&currentKey.toUpperCase()!==key.toUpperCase())
      throw Error('Frame Order '+id+' has a different existing POS item key. No rows changed.');
    if(!wasLinked&&!keyRow)
      throw Error('Frame Order '+id+' already exists. Cannot add the same frame to a different/new transaction.');
    return {row:found,isNew:false};
  }
  if(wasLinked)
    throw Error('Previously linked Frame Order '+id+' cannot be found. No duplicate row was created.');
  return {row:Math.max(last+1,2),isNew:true};
}

/** Write only changed cells: edits to an editing item must not rewrite an untouched frame. */
function posFrameSetChanged_(sh,row,col,value){
  const cell=sh.getRange(row,col);
  const existing=cell.getValue();
  const same=(typeof existing==='number'&&typeof value==='number')?
    Math.abs(existing-value)<0.000001:String(existing==null?'':existing)===String(value==null?'':value);
  if(!same)cell.setValue(value);
}

    else cell.clearContent();
  }
}
/** Write only truly changed cells, grouping consecutive columns into one call.
 * The original implementation used a read and possible write for EACH field,
 * which was especially slow for mixed editing + framing transactions.
 * Never touch unrelated cells, formulas, production status or blank columns.
 */
function posFrameWriteChangedBulkV1620_(sh,row,desired){
  const cols=Object.keys(desired).map(Number).sort((a,b)=>a-b);
  if(!cols.length)return;
  const current=sh.getRange(row,1,1,Math.max(...cols)).getValues()[0];
  const changes=[];
  for(const col of cols){
    const value=desired[col],prior=current[col-1];
    const equal=(typeof prior==='number'&&typeof value==='number')?
      Math.abs(prior-value)<0.000001:String(prior==null?'':prior)===String(value==null?'':value);
    if(!equal)changes.push({col,value});
  }
  let batch=[];
  const flush=()=>{
    if(!batch.length)return;
    sh.getRange(row,batch[0].col,1,batch.length).setValues([batch.map(x=>x.value)]);
    batch=[];
  };
  for(const item of changes){
    if(batch.length&&item.col!==batch[batch.length-1].col+1)flush();
    batch.push(item);
  }
  flush();
}

/** V16.18: Small, durable intake for NEW POS IDs.
 * Do not scan Tracking / REMIT / Frame Orders or run pricing lookups before
 * the ledger has acknowledged receipt. These checks happen in the worker.
 * A PENDING entry is NOT a completed sale.
 */
function posFastIntakeV1618_(r,proof,authorizedDeviceLabel){
  Logger.log('POS '+r.posId+' FAST_INTAKE_START');
  const seen=new Set();
  for(const item of r.items){
    const id=String(item.existingId||'').trim().toUpperCase();
    const pattern=item.kind==='editing'?/^RF\d{5}[A-Z]$/:/^[A-Z]{2}\d[A-Z]\d$/;
    if(!pattern.test(id))throw Error('Invalid '+item.kind+' order ID '+id+'. No payment recorded.');
    const key=item.kind+'|'+id;
    if(seen.has(key))throw Error('Duplicate order ID '+id+'. No payment recorded.');
    seen.add(key);item.existingId=id;
    if(item.kind==='frame'&&item.addOnType==='Personalized Letter')posLetterStaff_(item.addOnCommissionRecipient,true);
  }
  r.tipRecipient=String(r.tipRecipient||'Select').trim();
  r.received=posNum_(r.received||0);
  r.discount=posNum_(r.discount||0);
  r.tip=posNum_(r.tip||0);
  if(r.received<=0)throw Error('Received payment must be greater than zero. No payment recorded.');
  if(proof&&proof.uploadId){
    const saved=CacheService.getScriptCache().get('POS_PROOF_DONE_'+String(proof.uploadId));
    if(!saved)throw Error('Payment proof upload expired; upload again. No payment recorded.');
    const info=JSON.parse(saved);
    if(String(info.posId)!==String(r.posId))throw Error('Proof belongs to another POS ID.');
    r.proof=info.url;r.proofFileName=info.name;
  }
  if(r.proof&&!/^https:\/\//i.test(String(r.proof)))throw Error('Payment proof URL must be HTTPS.');
  if(r.items.some(x=>x.kind==='frame')&&!['LBC','LALAMOVE'].includes(r.shipping?.ship))throw Error('Choose LBC or LALAMOVE. No payment recorded.');
  r.admin=String(authorizedDeviceLabel||'Registered device');
  r.encodedAt=new Date().toISOString();r.updatedAt=r.encodedAt;
  // Received amount is authoritative; allocation and pricing are calculated by
  // the worker after ALL full validations. Do not use these values as synced sales.
  r.editPaid=0;r.framePaid=0;r.balance=0;
  r._fastIntakeV1618=true;
  if(!posWorkerInstalledV1616_())throw Error('Remittance worker missing. No payment recorded; run POS_V1616_INSTALL_WORKER as owner.');
  // Open ONLY the Master workbook, not the Tracking, REMIT or Frame Orders tabs.
  const ledger=posBook_().getSheetByName('POS Ledger');
  if(!ledger)throw Error('POS Ledger tab missing. No payment recorded.');
  const lock=LockService.getScriptLock();
  if(!lock.tryLock(8000))throw Error('POS is busy saving another order. Nothing confirmed; check POS Ledger by ID before retrying.');
  try{
    const found=posFindInCol_(ledger,1,r.posId,2);
    if(found>0){
      const prior=ledger.getRange(found,14,1,3).getValues()[0];
      const priorStatus=String(prior[0]||'').trim().toUpperCase();
      Logger.log('POS '+r.posId+' ALREADY_IN_LEDGER status='+priorStatus);
      // Do not overwrite it, even if a previous browser callback was lost.
      throw Error('POS #'+r.posId+' already exists in Ledger ('+priorStatus+'). Check Ledger; do not submit again.');
    }
    const row=Math.max(2,ledger.getLastRow()+1);
    if(row>ledger.getMaxRows())ledger.insertRowsAfter(ledger.getMaxRows(),Math.max(100,row-ledger.getMaxRows()));
    // Single write for all 17 fields, instead of repeatedly reformatting columns.
    ledger.getRange(row,1,1,17).setValues([[
      r.posId,r.paymentDate,r.encodedAt,r.updatedAt,
      String(r.first||'')+' '+String(r.last||''),r.email,r.admin,r.received,
      0,0,0,r.method||'',r.proof||'','PENDING','',JSON.stringify(r),0
    ]]);
    SpreadsheetApp.flush();
    Logger.log('POS '+r.posId+' FAST_INTAKE_COMMITTED row='+row);
    return {ok:true,queued:true,status:'PENDING',record:r,mode:'LIVE'};
  }finally{lock.releaseLock();}
}

/** Full server-side pricing and ID validations are intentionally deferred to
 * the scheduled worker, BEFORE any REMIT / Frame Orders / earnings write.
 * If a check fails, the ledger is marked ERROR and no auto-resubmission occurs.
 */
function posFinalizeQueuedV1618_(env,r,previous){
  Logger.log('POS '+r.posId+' WORKER_VALIDATE_START');
  posValidateIds_(env,r,previous||null);
  for(const item of r.items.filter(x=>x.kind==='frame'&&x.addOnType==='Personalized Letter')){
    const prior=(previous?.items||[]).find(x=>x.kind==='frame'&&x.existingId===item.existingId&&x.addOnType==='Personalized Letter');
    if(!prior||item.addOnCommissionRecipient)posLetterStaff_(item.addOnCommissionRecipient,true);
    if(prior?.addOnCommissionRecipient&&item.addOnCommissionRecipient!==prior.addOnCommissionRecipient)
      throw Error('Letter commission recipient cannot change after payment: '+item.existingId);
  }
  if(!posGetTipEmployees_().includes(String(r.tipRecipient||'Select')))throw Error('Tip recipient must be selected from REMIT column L.');
  if(r.items.some(x=>x.kind==='frame'))posFrameAddonEnsureHeaders_(env.frames);
  const cfg=rfFrameCfgRead_();
  for(const item of r.items.filter(x=>x.kind==='frame')){
    let kind;
    try{kind=rfFrameProductKind_(item.design);}catch(err){if(item.customSize)throw err;continue;}
    const quote=rfFrameProductQuote_(item,cfg);
    if(quote.needsQuote)throw Error('Configure '+item.design+' pricing before saving '+item.existingId+'.');
    if(quote.calculated&&Math.abs(Number(item.amount)-quote.price)>0.01)throw Error('Product price changed; refresh calculator: '+item.existingId+'.');
    if(item.customSize)item.size=Number(item.customWidth)+'"x'+Number(item.customHeight)+'" - Custom '+(kind==='framed'?'Black Frame':kind==='print'?'Print Only':'Canvas Only');
  }
  const freight=rfFrameShippingQuote_(r,cfg);
  r.shippingFee=freight.fee;r.shippingTier=freight.tier;r.shippingDays=freight.days;r.shippingManual=freight.manual;
  const subtotal=r.items.reduce((a,x)=>a+Number(x.amount)+posFrameAddonCharge_(x),0);
  r.discount=posNum_(r.discount||0);r.tip=posNum_(r.tip||0);r.received=posNum_(r.received||0);
  if(r.discount>subtotal)throw Error('Discount exceeds subtotal.');
  const actualDue=posMoney_(subtotal-r.discount+r.tip+r.shippingFee);
  if(r.received+0.001<actualDue)throw Error('Amount paid below actual amount due. Check order/payment before retrying.');
  r.excess=posMoney_(Math.max(0,r.received-actualDue));
  const ed=r.items.filter(x=>x.kind==='editing').reduce((a,x)=>a+Number(x.amount),0);
  const net=subtotal-r.discount+r.shippingFee;
  const paidBeforeTip=Math.max(0,Math.min(r.received,actualDue)-r.tip);
  r.editPaid=net?posMoney_(paidBeforeTip*(ed-(subtotal?r.discount*ed/subtotal:0))/net):0;
  r.framePaid=posMoney_(paidBeforeTip-r.editPaid);r.balance=0;r.due=actualDue;
  posServiceAmounts_(r);
  Logger.log('POS '+r.posId+' WORKER_VALIDATED');
}

function posSaveTransaction_(input,proof,authorizedDeviceLabel){
  if(!input||!/^[0-9]{8}$/.test(String(input.posId||'')))throw Error('Invalid POS transaction ID.');
  const r=JSON.parse(JSON.stringify(input));posDate_(r.paymentDate);
  if(!String(r.first||'').trim()||!String(r.email||'').trim()||!Array.isArray(r.items)||!r.items.length)throw Error('Customer name (one word or N/A), email or N/A, and at least one service are required.');
  r.email=String(r.email||'').trim();
  if(/^n\/?a$/i.test(r.email)){r.email='N/A';}
  else if(!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(r.email))throw Error('Enter a valid email address or choose N/A (email pending).');
  r.shipping=r.shipping&&typeof r.shipping==='object'?r.shipping:{};
  r.shipping.contactPhone=String(r.shipping.contactPhone||'').trim().slice(0,50);
  if(r.items.length>20)throw Error('Maximum 20 services per transaction.');
  if(r.items.some(x=>!['editing','frame'].includes(x.kind)))throw Error('Unknown service type.');
  const frameIds=new Set(r.items.filter(x=>x.kind==='frame').map(x=>String(x.existingId||'').trim().toUpperCase()));
  const allocatedFrames=new Set();
  for(const item of r.items){
    item.amount=posNum_(item.amount);
    if(item.kind==='frame')posFrameAddonValidate_(item);
    if(item.kind!=='editing')continue;
    item.frameOrderId=String(item.frameOrderId||'').trim().toUpperCase();
    if(item.frameOrderId){
      if(!frameIds.has(item.frameOrderId))throw Error('Frame ID '+item.frameOrderId+' is not in this transaction.');
      if(allocatedFrames.has(item.frameOrderId))throw Error('Frame ID '+item.frameOrderId+' is assigned to multiple editing items.');
      allocatedFrames.add(item.frameOrderId);
    }
    if(item.fileType==='RUSH FILE'){
      if(!item.rushTargetDate)throw Error('Target date is required for rush editing '+item.existingId+'.');
      posDate_(item.rushTargetDate);
    }else item.rushTargetDate='';
  }
  // V16.20: both NEW transactions and EXISTING transaction edits use
  // a short, durable Ledger intake. The owner worker performs slow validations
  // and multi-sheet writes after the browser has a receipt for the POS ID.
  if(!r.updatedAt)return posFastIntakeV1618_(r,proof,authorizedDeviceLabel);
  return posFastEditIntakeV1620_(r,proof,authorizedDeviceLabel);
}

/** V16.20: edit intake opens ONLY the POS Ledger; no REMIT / Frame Orders
 * scans, pricing lookups or expensive sheet writes under the save lock.
 * The previous ledger snapshot stays embedded for safe row reconciliation.
 * This is a QUEUED UPDATE, not a new customer payment.
 */
function posFastEditIntakeV1620_(r,proof,authorizedDeviceLabel){
  const previousVersion=String(r.updatedAt||'');
  if(!previousVersion)throw Error('Existing-order update requires the original version.');
  r.tipRecipient=String(r.tipRecipient||'Select').trim();
  r.received=posNum_(r.received||0);
  r.tip=posNum_(r.tip||0);
  r.discount=posNum_(r.discount||0);
  if(proof&&proof.uploadId){
    const saved=CacheService.getScriptCache().get('POS_PROOF_DONE_'+String(proof.uploadId));
    if(!saved)throw Error('Payment proof upload expired; please upload it again. Nothing changed.');
    const info=JSON.parse(saved);
    if(String(info.posId)!==String(r.posId))throw Error('Proof belongs to a different POS ID.');
    r.proof=info.url;r.proofFileName=info.name;
  }
  if(r.proof&&!/^https:\/\//i.test(String(r.proof)))throw Error('Proof must be a secure HTTPS URL.');
  if(r.items.some(x=>x.kind==='frame')&&!['LBC','LALAMOVE'].includes(r.shipping?.ship))
    throw Error('Choose LBC or LALAMOVE. Nothing changed.');
  if(!posWorkerInstalledV1616_())throw Error('POS background worker not installed; nothing changed.');
  const ledger=posBook_().getSheetByName('POS Ledger');
  if(!ledger)throw Error('POS Ledger missing; nothing changed.');
  const lock=LockService.getScriptLock();
  if(!lock.tryLock(8000))throw Error('Another POS update is being recorded. Check Ledger before retrying.');
  try{
    const row=posFindInCol_(ledger,1,r.posId,2);
    if(row<0)throw Error('Existing POS ID not found. Refresh the transaction; nothing changed.');
    const oldCells=ledger.getRange(row,14,1,3).getValues()[0];
    const status=String(oldCells[0]||'').trim().toUpperCase();
    if(status!=='SYNCED')throw Error('POS #'+r.posId+' is '+status+'. Do not submit another update; check Ledger.');
    const old=JSON.parse(oldCells[2]);
    if(old.deletedAt)throw Error('Deleted transactions cannot be changed.');
    if(previousVersion!==String(old.updatedAt))throw Error('Transaction changed elsewhere. Refresh it before editing.');
    const original=(old.items||[]).map(x=>x.kind+'|'+String(x.existingId||'').toUpperCase());
    const incoming=(r.items||[]).map(x=>x.kind+'|'+String(x.existingId||'').toUpperCase());
    if(original.some(key=>!incoming.includes(key)))throw Error('Linked order IDs cannot be removed or changed.');
    // Stash pre-edit snapshot in the pending record for safe existing-row lookup.
    r._queuedEditPreviousV1620=old;
    r._fastIntakeV1618=true;
    r._queuedEditV1620=true;
    r.encodedAt=old.encodedAt;
    r.updatedAt=new Date().toISOString();
    r.admin=String(authorizedDeviceLabel||'Registered device');
    // Preserve old allocation until the worker recalculates everything.
    r.editPaid=Number(old.editPaid||0);r.framePaid=Number(old.framePaid||0);
    r.balance=Number(old.balance||0);r.excess=Number(old.excess||0);
    const values=[r.posId,r.paymentDate,r.encodedAt,r.updatedAt,
      String(r.first||'')+' '+String(r.last||''),r.email,r.admin,r.received,
      r.editPaid,r.framePaid,r.balance,r.method||'',r.proof||'',
      'PENDING','EDIT QUEUED — previous record preserved',JSON.stringify(r),r.excess];
    // One 17-column write under the lock. Never append an existing transaction.
    ledger.getRange(row,1,1,17).setValues([values]);
    SpreadsheetApp.flush();
    Logger.log('POS '+r.posId+' FAST_EDIT_QUEUED row='+row);
    return {ok:true,queued:true,status:'PENDING',record:r,mode:'LIVE',edit:true};
  }finally{lock.releaseLock();}
}

/** Resumable Drive upload for JPEG / PNG / WEBP / PDF, up to 25 MiB. */
function posBeginProofUpload_(posId,name,mimeType,size,smallBase64){
  const allowed=['image/jpeg','image/png','image/webp','application/pdf'];
  if(!/^\d{8}$/.test(String(posId)))throw Error('Invalid POS ID.');
  if(!allowed.includes(mimeType))throw Error('Only JPG, PNG, WEBP and PDF are accepted.');
  if(!Number.isInteger(size)||size<1||size>25*1024*1024)throw Error('Maximum payment proof is 25 MB.');
  const folderId=PropertiesService.getScriptProperties().getProperty(POS_FOLDER_PROP);
  if(!folderId)throw Error('Payment proof folder is not configured.');
  const safe=String(name||'proof').replace(/[^a-zA-Z0-9_.-]/g,'_').slice(0,100);
  // V16.21: Small receipt, single authorized route call + one Drive upload.
  // No duplicate path in REMIT; 'save' later consumes a short-lived proof token.
  if(smallBase64!==undefined&&smallBase64!==null){
    if(typeof smallBase64!=='string'||!smallBase64||size>900*1024)
      throw Error('Invalid compressed proof payload.');
    const bytes=Utilities.base64Decode(smallBase64);
    if(bytes.length!==size)throw Error('Proof payload size mismatch; upload was not accepted.');
    const blob=Utilities.newBlob(bytes,mimeType,posId+'_'+safe);
    const folder=DriveApp.getFolderById(folderId);
    const uploaded=folder.createFile(blob);
    const token=Utilities.getUuid().replace(/-/g,'');
    const url='https://drive.google.com/file/d/'+uploaded.getId()+'/view';
    CacheService.getScriptCache().put('POS_PROOF_DONE_'+token,JSON.stringify({url,name:safe,posId}),1200);
    return {uploadId:token,done:true,url};
  }

  const res=UrlFetchApp.fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable&fields=id,webViewLink,name',{method:'post',muteHttpExceptions:true,contentType:'application/json',headers:{Authorization:'Bearer '+ScriptApp.getOAuthToken(),'X-Upload-Content-Type':mimeType,'X-Upload-Content-Length':String(size)},payload:JSON.stringify({name:posId+'_'+safe,parents:[folderId]})});
  if(![200,201].includes(res.getResponseCode()))throw Error('Cannot initialize Drive upload: '+res.getResponseCode()+' '+res.getContentText().slice(0,160));
  const session=res.getHeaders().Location||res.getHeaders().location;if(!session)throw Error('Missing Drive upload session.');
  const token=Utilities.getUuid().replace(/-/g,'');CacheService.getScriptCache().put('POS_PROOF_SESSION_'+token,JSON.stringify({session,size,posId,name:safe,mimeType,next:0}),1200);
  return {uploadId:token,chunkSize:2*1024*1024};
}
function posUploadProofChunk_(uploadId,base64,start){
  posRequireAuth_();const cache=CacheService.getScriptCache(),key='POS_PROOF_SESSION_'+String(uploadId),raw=cache.get(key);if(!raw)throw Error('Upload session expired; try again.');
  const info=JSON.parse(raw),bytes=Utilities.base64Decode(base64);if(start!==info.next||bytes.length<1||bytes.length>2*1024*1024||start+bytes.length>info.size)throw Error('Invalid upload chunk.');
  const end=start+bytes.length-1;
  const res=UrlFetchApp.fetch(info.session,{method:'put',muteHttpExceptions:true,contentType:info.mimeType,payload:bytes,headers:{Authorization:'Bearer '+ScriptApp.getOAuthToken(),'Content-Range':'bytes '+start+'-'+end+'/'+info.size}});
  const code=res.getResponseCode();
  if(end+1<info.size){if(code!==308)throw Error('Upload failed at '+start+': HTTP '+code);info.next=end+1;cache.put(key,JSON.stringify(info),1200);return {done:false,uploaded:info.next,total:info.size};}
  if(code!==200&&code!==201)throw Error('Upload completion failed: HTTP '+code+' '+res.getContentText().slice(0,180));
  const file=JSON.parse(res.getContentText()),url=file.webViewLink||'https://drive.google.com/file/d/'+file.id+'/view';cache.put('POS_PROOF_DONE_'+uploadId,JSON.stringify({url,name:info.name,posId:info.posId}),1200);cache.remove(key);return {done:true,url};
}

/** V11 — Clear the complete Tracking row for a DELETED POS editing item.
 * Every target is identified by the exact persisted editing ID, and only POS-era
 * rows (10251 onward) may be cleared. No row is physically deleted: sheet
 * structure, validation and formatting stay intact. Other orders are untouched.
 */
function posClearDeletedTrackingDetails_(record){
  const ids=new Set((record.items||[]).filter(x=>x.kind==='editing')
    .map(x=>String(x.existingId||'').trim().toUpperCase()).filter(Boolean));
  if(!ids.size)return 0;
  const tr=posBook_().getSheetByName('Tracking');
  if(!tr)throw Error('Tracking sheet not found for deleted POS order cleanup.');
  const first=10251,last=tr.getLastRow();if(last<first)return 0;
  const values=tr.getRange(first,1,last-first+1,1).getDisplayValues();
  const candidates=[];
  values.forEach((r,i)=>{if(ids.has(String(r[0]||'').trim().toUpperCase()))candidates.push({row:first+i,id:String(r[0]).trim().toUpperCase()});});
  const seen=new Set();for(const x of candidates){
    if(seen.has(x.id))throw Error('Duplicate Tracking ID '+x.id+'; manual cleanup required.');
    seen.add(x.id);
  }
  let count=0;
  for(const x of candidates){
    // Recheck just before editing in case the Tracking row moved.
    if(String(tr.getRange(x.row,1).getDisplayValue()||'').trim().toUpperCase()!==x.id)
      throw Error('Tracking order ID changed before cleanup: '+x.id);
    tr.getRange(x.row,1,1,tr.getMaxColumns()).clearContent();
    count++;
  }
  return count;
}

function posDeleteTransaction_(posId, expectedUpdatedAt,authorizedDeviceLabel){
  posRequireAuth_();
  if(!/^\d{8}$/.test(String(posId)))throw Error('Invalid POS transaction ID.');
  const lock=LockService.getScriptLock();lock.waitLock(30000);
  try{
    const env=posEnv_();
    const ledgerRow=posFindInCol_(env.ledger,1,posId,2);
    if(ledgerRow<2)throw Error('POS transaction not found.');
    const current=env.ledger.getRange(ledgerRow,14,1,3).getValues()[0];
    const state=String(current[0]||'');
    const record=JSON.parse(current[2]);
    if(state==='DELETED'||record.deletedAt){
      // Even after a prior successful deletion, finish any leftover Tracking cleanup.
      const cleared=posClearDeletedTrackingDetails_(record);
      return {ok:true,posId,alreadyDeleted:true,trackingRowsCleared:cleared};
    }
    if(state!=='SYNCED'&&state!=='DELETING')throw Error('This transaction is not fully synced. Repair the sync before deleting.');
    if(state!=='DELETING'&&String(record.updatedAt)!==String(expectedUpdatedAt))throw Error('Transaction changed. Refresh history before deleting.');
    const targets=[];
    for(const item of record.items||[]){
      const sh=item.kind==='editing'?env.remit:env.frames;
      const col=item.kind==='editing'?18:40;
      const first=item.kind==='editing'?6:2;
      const row=posFindInCol_(sh,col,posItemKey_(record,item),first);
      if(row<0){
        // The linked sheet row may already have been manually removed.
        // Still archive and mark the POS Ledger transaction DELETED.
        continue;
      }
      if(item.kind==='editing'){
        const vals=sh.getRange(row,16,1,3).getDisplayValues()[0];
        if(String(vals[0]).trim()!==String(posId)||String(vals[2]).trim()!==posItemKey_(record,item))throw Error('REMIT row linkage mismatch for '+item.existingId);
        const actual=String(sh.getRange(row,17).getDisplayValue()).trim();
        if(actual!==String(item.existingId))throw Error('Editing ID changed for '+item.existingId+'; manual review required.');
      } else {
        const status=String(sh.getRange(row,20).getDisplayValue()||'').trim().toUpperCase();
        if(status!=='HOLD')throw Error('Frame '+item.existingId+' status is '+status+'. Only HOLD frame orders can be deleted; contact production first.');
        const workflow=sh.getRange(row,21,1,18).getDisplayValues()[0];
        if(workflow.some(x=>String(x).trim()))throw Error('Frame '+item.existingId+' has email, mockup, tracking or production information. Deletion blocked.');
        if(String(sh.getRange(row,3).getDisplayValue()).trim()!==String(item.existingId))throw Error('Frame order ID mismatch for '+item.existingId);
      }
      targets.push({sh,row,kind:item.kind,id:item.existingId});
    }
    const archive=posEnsureSheet_(posBook_(),'POS Deleted Archive',['Deleted At','POS ID','Deleted By','Original Payment Date','Amount Received','Original Record JSON']);
    const existing=posFindInCol_(archive,2,posId,2);
    const now=new Date().toISOString(),admin=String(authorizedDeviceLabel || 'Registered device');
    if(existing<0)archive.appendRow([now,posId,admin,record.paymentDate,record.received,JSON.stringify(record)]);
    env.ledger.getRange(ledgerRow,14,1,2).setValues([['DELETING','Removing POS-linked orders']]);
    // No row deletion: preserve formatted blank templates and downstream row references.
    for(const t of targets){
      if(t.kind==='editing'){
        t.sh.getRange(t.row,1,1,6).clearContent();
        t.sh.getRange(t.row,7,1,4).clearContent();  // G:J file smart chips / URLs (V11 FIX)
        t.sh.getRange(t.row,11,1,16).clearContent(); // K:Z
        t.sh.getRange(t.row,27).clearContent(); // AA: POS rush target date
      }else{
        t.sh.getRange(t.row,1,1,20).clearContent(); // A:T
        t.sh.getRange(t.row,39,1,9).clearContent(); // AM:AU
      }
    }
    record.deletedAt=now;record.deletedBy=admin;record.updatedAt=now;
    env.ledger.getRange(ledgerRow,4).setValue(now);
    env.ledger.getRange(ledgerRow,14,1,3).setValues([['DELETED','',JSON.stringify(record)]]);
    env.audit.appendRow([now,'DELETE',posId,admin,record.paymentDate,'',record.received,0,'DELETED','']);
    let trackingRowsCleared=0,trackingCleanupError='';
    try{trackingRowsCleared=posClearDeletedTrackingDetails_(record);}catch(e){
      trackingCleanupError=String(e.message||e);
      Logger.log('Deleted POS '+posId+' Tracking cleanup pending: '+trackingCleanupError);
    }
    // A partial cleanup must be visible to the cashier, not silently described as complete.
    return {ok:!trackingCleanupError,posId,removed:targets.length,trackingRowsCleared,
      trackingCleanupError,deleted:true,needsAttention:!!trackingCleanupError};
  }finally{lock.releaseLock();}
}

function rfLateAddonViberNotify_(text,id){

  const ids=log.getRange(2,2,n,1).getDisplayValues();

   const v=log.getRange(i+2,3,1,4).getDisplayValues()[0];

/** V16.12: reconcile an existing paid add-on from its original payment log.
 * Never create another payment, never send another Viber report.
 * Reject conflict instead of overwriting any different existing add-on.
 */
function rfLateAddonReconcile_(id){
 id=rfLateAddonId_(id);
 const prev=rfLateAddonExistingPayment_(id);
 if(!prev)throw Error('No recorded late add-on payment for '+id+'. No repair performed.');
 const v=prev.sheet.getRange(prev.row,1,1,16).getDisplayValues()[0];
 const paid=Number(String(v[5]||'').replace(/[^\d.-]/g,''));
 if(paid!==200||String(v[1]||'').trim().toUpperCase()!==id)throw Error('Payment record mismatch. Manual review required; no changes made.');
 const item={kind:'frame',existingId:id,addOnType:'Personalized Letter',addOnColor:String(v[8]||''),addOnFont:String(v[9]||''),addOnMessage:String(v[10]||'')};
 posFrameAddonValidate_(item);
 const sh=posSheet_(POS_FRAME_ID,'Orders'),row=posFindInCol_(sh,3,id,2);
 if(row<2)throw Error('Frame order not found. No changes made.');
 const current=sh.getRange(row,48,1,5).getDisplayValues()[0];
 const target=['Personalized Letter','200',item.addOnFont,item.addOnMessage,item.addOnColor];
 const matches=(v,i)=>i===1?Number(String(v||'').replace(/[^\d.-]/g,''))===200:String(v||'').trim()===target[i];
 const conflicts=current.map((v,i)=>String(v||'').trim()&&!matches(v,i)?' '+['AV','AW','AX','AY','AZ'][i]:null).filter(Boolean);
 if(conflicts.length)throw Error('Frame add-on fields conflict with the original paid record at '+conflicts.join(',')+'. No overwrite performed.');
 let restored=false;
 if(current.some((v,i)=>!matches(v,i))){
   rfLateAddonWriteFrame_(sh,row,item);
   SpreadsheetApp.flush();restored=true;
 }

/** V16.16 — asynchronous two-phase remittance. Never resend a payment to retry.
 * Runs only in the POS script project, not the original Viber Channel project.
 * Install ONCE as the script owner using POS_V1616_INSTALL_WORKER(). */
const POS_V1616_WORKER='posProcessQueuedRemittancesV1616';
function posWorkerInstalledV1616_(){
  // Trigger enumeration is comparatively expensive. Cache only for a short window.
  const cache=CacheService.getScriptCache(),key='RF_POS_V1617_WORKER_PRESENT';
  if(cache.get(key)==='1')return true;
  const installed=ScriptApp.getProjectTriggers().some(t=>t.getHandlerFunction()===POS_V1616_WORKER && t.getEventType()===ScriptApp.EventType.CLOCK);
  if(installed)cache.put(key,'1',120);
  return installed;
}

  const caller=String(Session.getActiveUser().getEmail()||'').trim().toLowerCase();
  const owner=String(Session.getEffectiveUser().getEmail()||'').trim().toLowerCase();
  if(caller && owner && caller!==owner)throw Error('Run as the Apps Script owner.');
  const all=ScriptApp.getProjectTriggers().filter(t=>t.getHandlerFunction()===POS_V1616_WORKER);
  if(all.length>1)throw Error('Multiple V16.16 worker triggers exist. Inspect triggers before making changes.');
  if(!all.length)ScriptApp.newTrigger(POS_V1616_WORKER).timeBased().everyMinutes(1).create();
  const result={installed:posWorkerInstalledV1616_(),handler:POS_V1616_WORKER,duplicates:0,note:'This worker is independent of the original Viber Bridge.'}; Logger.log(JSON.stringify(result));return result;

/**
 * V16.18: lock is held ONLY while claiming a pending transaction.
 * All slow REMIT, Frame Orders and earnings operations happen OUTSIDE the lock.
 * The claim is durable (POS Ledger column N = PROCESSING) so concurrent minute
 * triggers cannot pick up the same POS ID. Failed partial sync => ERROR,
 * never automatically resubmit or send again.
 *
 * Existing trigger handler name is retained for zero-downtime V16.16 upgrades.
 */
/** V16.20: If an editor changes only the editing portion of a mixed
 * transaction, do NOT touch its unchanged frame row. Compare service data
 * and the transaction fields that are actually projected into that service.
 */
function posShouldWriteEditedItemV1620_(r,item){
  const prev=r._queuedEditPreviousV1620;
  if(!prev)return true;
  const earlier=(prev.items||[]).find(x=>x.kind===item.kind&&x.existingId===item.existingId);
  if(!earlier)return true;
  const clean=x=>{const c=JSON.parse(JSON.stringify(x));delete c._addonOpen;delete c.uid;return c;};
  if(JSON.stringify(clean(item))!==JSON.stringify(clean(earlier)))return true;
  const fields=['paymentDate','method','first','last','email','phone','proof','remarks','tipRecipient','tip','discount','received'];
  if(fields.some(k=>String(r[k]??'')!==String(prev[k]??'')))return true;
  return JSON.stringify(r.shipping||{})!==JSON.stringify(prev.shipping||{});
}

function posProcessQueuedRemittancesV1616(){
  const book=posBook_(),ledger=book.getSheetByName('POS Ledger');
  if(!ledger||ledger.getLastRow()<2)return;
  const snapshot=ledger.getRange(2,1,ledger.getLastRow()-1,14).getDisplayValues();
  const started=Date.now();let claimed=0;
  for(let i=0;i<snapshot.length;i++){
    if(claimed>=2||Date.now()-started>210000)break;
    if(String(snapshot[i][13]||'').trim().toUpperCase()!=='PENDING')continue;
    const row=i+2;
    const lock=LockService.getScriptLock();
    if(!lock.tryLock(1500))return; // another short save/claim; next minute will retry
    let record=null,id='';
    try{
      if(String(ledger.getRange(row,14).getDisplayValue()).trim().toUpperCase()!=='PENDING')continue;
      id=String(ledger.getRange(row,1).getDisplayValue()).trim();
      record=JSON.parse(ledger.getRange(row,16).getValue());
      if(String(record.posId)!==id)throw Error('Ledger POS ID and JSON do not match.');
      const claimTime=new Date().toISOString();
      ledger.getRange(row,14,1,2).setValues([['PROCESSING','Claimed '+claimTime]]);
      SpreadsheetApp.flush();
    }catch(e){
      const reason=String(e&&e.message||e).slice(0,300);
      ledger.getRange(row,14,1,2).setValues([['ERROR',reason]]);
      console.error('POS queue claim row '+row+': '+reason);
      record=null;
    }finally{lock.releaseLock();}
    if(!record)continue;
    claimed++;
    try{
      Logger.log('POS '+id+' SYNC_START row '+row);
      const env=posEnv_();
      if(record._fastIntakeV1618){
        // Validate completely before any downstream write, then preserve the
        // finalized allocations in Ledger while still marked PROCESSING.
        posFinalizeQueuedV1618_(env,record,record._queuedEditPreviousV1620||null);
        posWriteLedger_(env,record,'PROCESSING','Full validation passed',row);
        SpreadsheetApp.flush();
      }else{
        posServiceAmounts_(record);
      }
      for(const item of record.items){
        if(!posShouldWriteEditedItemV1620_(record,item)){
          Logger.log('POS '+id+' UNCHANGED_ITEM_SKIPPED '+item.kind+' '+item.existingId);
          continue;
        }
        if(item.kind==='editing')posSyncRemit_(env,record,item);
        else posSyncFrame_(env,record,item,record._queuedEditPreviousV1620||record);
      }
      posRecordEarnings_(record,posEarningsSheet_());
      // Do not overwrite a row changed manually while this worker was running.
      const state=String(ledger.getRange(row,14).getDisplayValue()).trim().toUpperCase();
      if(state!=='PROCESSING')throw Error('Ledger sync state changed while processing: '+state);
      delete record._queuedEditPreviousV1620;
      delete record._queuedEditV1620;
      posWriteLedger_(env,record,'SYNCED','',row);
      SpreadsheetApp.flush();
      Logger.log('POS '+id+' SYNCED');
      try{env.audit.appendRow([new Date().toISOString(),'QUEUE_SYNC',id,record.admin,'','','',record.received,'SYNCED','']);}
      catch(auditErr){console.error('POS '+id+' synced; audit log warning: '+String(auditErr));}
    }catch(e){
      const reason=String(e&&e.message||e).slice(0,300);
      // If this attempt partially wrote downstream sheets, do not retry automatically.
      const state=String(ledger.getRange(row,14).getDisplayValue()).trim().toUpperCase();
      if(state==='PROCESSING')ledger.getRange(row,14,1,2).setValues([['ERROR',reason]]);
      try{book.getSheetByName('POS Audit').appendRow([new Date().toISOString(),'QUEUE_SYNC',id,'Worker','','','','','ERROR',reason]);}
      catch(auditErr){console.error('Audit write warning: '+String(auditErr));}
      console.error('POS '+id+' requires review: '+reason);
    }
  }
}
